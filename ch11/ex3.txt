Fixed:
void avg_sum(double a[], int n, double *avg, double *sum)
{
  int i;
  sum = 0.0;
  for (i = 0, i < n, i++) {
    *sum += a[i]; // we want to store result in what "sum" points to, not in sum
  }
  *avg = *sum / n; // we want to store result in what "avg" points to, and use the value stored in what "sum" points to for calculation
}
