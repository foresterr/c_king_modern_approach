int f(char *s, char *t)
{
  char *p1, *p2;
  for (p1 = s; *p1; p1++) {
    for (p2 = t; *p2; p2++)
      if (*p1 == *p2) break;
    if (*p2 == '\0') break;
  }
  return p1 - s;
}

f("abcd", "babc") = 3
f("abcd", "bcd") = 0

This will return position (zero based) of first character in s that does not occur in t, or length of s + 1 if all characters of s occur in t
