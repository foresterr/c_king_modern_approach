Original:
int find_largest(int a[], int n)
{
  int i, max;
  max = a[0];
  for (i = 1; i < n; i++)
    if (a[i] > max)
      max = a[i];
  return max;
}

Rewritten:
int find_largest(int *a, int n)
{
  max = *a;
  for (int *p = a + 1; p < a + n; p++)
    if (*p > max)
      max = *p;
  return max;
}
