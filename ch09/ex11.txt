float compute_GPA(char grades[], int n)
{
  int total = 0;
  for (int i = 0; i < n; i++) {
    switch (grades[i]) {
      case 'A': case 'a':
        total += 4;
        break;
      case 'B': case 'b':
        total += 3;
        break;
      case 'C': case 'c':
        total += 2;
        break;
      case 'D': case 'd':
        total += 1;
        break;
      //F is 0, so we don't need a case for it
    }
  }
  return (float) total / n;
}
