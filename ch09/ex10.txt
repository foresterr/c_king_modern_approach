int largest(int array[], int n)
{
  int largest = array[0];
  for (int i = 1; i < n; i++) {
    if (array[i] > largest)
      largest = array[i];
  }
  return largest;
}

double average(int array[], int n)
{
  int sum = 0;
  for (int i = 0; i < n; i++) {
    sum += array[i];
  }
  return (double) sum / n;
}

int positive_count(int array[], int n)
{
  int count = 0;
  for (int i = 0; i < n; i++) {
    if (array[i] > 0)
      count++;
  }
  return count;
}
