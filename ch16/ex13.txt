struct shape {
  int shape_kind;      /* RECTANGLE or CIRCLE */
  struct point center; /* coordinates of center */
  union {
    struct {
      int height, width;
    } rectangle;
    struct {
      int radius;
    } circle;
  } u;
} s;

s.shape_kind = RECTANGLE; legal only if appropriate enum or macro constants are defined before
s.center.x = 10; legal
s.height = 25; illegal, should be s.u.rectangle.height = 25
s.u.rectangle.width = 8; legal
s.u.circle = 5; illegal, should be s.u.circle.radius = 5
s.u.radius = 5; illegal, should be s.u.circle.radius = 5
