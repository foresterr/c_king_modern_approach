#include <stdio.h>

#define N 100

void f(void);

int main(void)
{
  f();
#ifdef N
#undef N
#endif
  return 0;
}

void f(void)
{
#if defined(N)
  printf("N is %d\n", N);
#else
  printf("N is undefined\n");
#endif
}

After preprocessing:


void f(void);

int main(void)
{
  f();



  return 0;
}

void f(void)
{



  printf("N is undefined\n");

}

It will print "N is undefined". Preprocessor directives are processed from top to bottom of the file,
not during function call
